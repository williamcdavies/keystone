You are given two **non-empty** linked lists representing two non-negative integers. The digits are stored in **reverse order**, and each of their nodes contains a single digit. Add the two numbers and return the sum as a linked list.

```cpp
/**
* Definition for singly-linked list.
* struct ListNode {
	* int val;
	* ListNode *next;
	* ListNode() : val(0), next(nullptr) {}
	* ListNode(int x) : val(x), next(nullptr) {}
	* ListNode(int x, ListNode *next) : val(x), next(next) {}
* };
*/

class Solution {
	public:
		ListNode* addTwoNumbers(ListNode* l1, 
								ListNode* l2) {
			ListNode* res = new ListNode();
			ListNode* it  = res;
			int       car = 0;
			
			while (l1 != nullptr 
				   || l2 != nullptr) {
				ListNode* new_node = new ListNode();
				int       sum      = 0;
				int       v1       = 0;
				int       v2       = 0;
				
				it->next = new_node;
				it       = it->next;
				
				if (l1 != nullptr) v1 = l1->val
				if (l2 != nullptr) v2 = l2->val;
				
				sum           = v1 
							  + v2 
						      + car;
				new_node->val = sum % 10;
				
				if (sum > 9) car = sum - 10;
				else         car = 0
				
				if (l1 != nullptr) l1 = l1->next;
				if (l2 != nullptr) l2 = l2->next;
			}
			
			return res;
	}
};
```

- we have to consider the length of each l1 and l2. if one is longer than the other we could hit an error -> can't base loop of len of l1 or l2. 
- what we want to do: two iterators. if one iterator hits end, stop. continue other.
