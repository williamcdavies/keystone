Given a string `s`, find the length of the **longest** without duplicate characters.

```cpp
class Solution {
	public:
		int lengthOfLongestSubstring(string s) {
			unordered_map<char, size_t> map;
			size_t                      res = 0;
			size_t                      l   = 0;
			size_t                      r   = 0;
			
			while (r < s.length()) {
				char key = s[r];
				
				while (map.count(key)) {
					map.erase(s[l]);
					++l;
				}
				
				map.insert({key, r});
				++r;
				
				res = max(res, r - l);
			}
			
			return res;
		}
};
```

> [!note]
> `while(map.count(key))` can be used to replace `for(auto search = map.find(key); search != map.end(); search = map.find(key))`.

> [!note]
> `if (b > a) a = b` and `a = max(a, b)` are equivalent.
